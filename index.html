Oracle SQL Query

SELECT
    'SELECT ''' || column_name || ''' AS column_name, ' ||
    'COUNT(*) AS total_rows, ' ||
    'COUNT(' || column_name || ') AS non_null_count, ' ||
    'COUNT(*) - COUNT(' || column_name || ') AS null_count, ' ||
    'ROUND(COUNT(' || column_name || ') / COUNT(*) * 100, 2) AS fill_rate_pct ' ||
    'FROM MC_NBE_RETAIL_CUSTOMER_DEMOGRAPHICS UNION ALL'
FROM user_tab_columns
WHERE table_name = 'MC_NBE_RETAIL_CUSTOMER_DEMOGRAPHICS'
ORDER BY column_id;
